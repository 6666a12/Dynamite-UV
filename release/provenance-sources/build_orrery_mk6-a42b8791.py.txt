def mat_pbr(name, base, metallic, roughness, emission=None, strength=0.0,
            spec=0.5):
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]

    def put(key, value):
        sock = bsdf.inputs.get(key)
        if sock is not None:
            sock.default_value = value

    put("Base Color", (*base, 1.0))
    put("Metallic", metallic)
    put("Roughness", roughness)
    put("Specular IOR Level", spec)
    if emission is not None:
        put("Emission Color", (*emission, 1.0))
        put("Emission Strength", strength)
    mat.use_backface_culling = True
    return mat

def mesh_object(name, verts, faces, mat):
    me = bpy.data.meshes.new(name)
    me.from_pydata(verts, [], faces)
    me.validate(verbose=False)
    bm = bmesh.new()
    bm.from_mesh(me)
    bmesh.ops.recalc_face_normals(bm, faces=bm.faces)
    bm.to_mesh(me)
    bm.free()
    me.update()
    ob = bpy.data.objects.new(name, me)
    bpy.context.collection.objects.link(ob)
    me.materials.append(mat)
    return ob


def empty(name, parent=None, loc=(0, 0, 0), rot=(0, 0, 0)):
    ob = bpy.data.objects.new(name, None)
    ob.empty_display_size = 0.4
    bpy.context.collection.objects.link(ob)
    ob.parent = parent
    ob.location = loc
    ob.rotation_mode = "XYZ"
    ob.rotation_euler = rot
    return ob


# Ten-section folded beam: low crest, one bevel step per side, sharp waist,
# flat back. The front breaks into planes instead of reading as one card.
PROFILE = ((0.00, -1.00), (0.55, -0.72), (0.88, -0.22), (1.00, 0.30),
           (0.55, 0.86), (0.00, 1.00), (-0.55, 0.86), (-1.00, 0.30),
           (-0.88, -0.22), (-0.55, -0.72))


def taper(u, edge, floor):
    s = min(u, 1.0 - u) / edge
    return floor + (1.0 - floor) * min(1.0, max(0.0, s))


def loft_band(name, pts, half_w, half_t, mat, crest=(0.0, 0.0, 1.0),
              edge=0.14, floor=0.5, twist=0.0, closed=False, prof=PROFILE,
              bevel_mat=None):
    """Loft a chamfered rail along `pts`.

    `crest` is the direction the ridge faces (the plane normal here), `twist`
    rotates the section about the path, and `bevel_mat` paints the four front
    facets so the cut reads by value as well as by light.
    """
    crest_v = Vector(crest).normalized()
    n = len(pts)
    tangents = []
    for i in range(n):
        if closed:
            a, b = pts[(i - 1) % n], pts[(i + 1) % n]
        else:
            a, b = pts[max(i - 1, 0)], pts[min(i + 1, n - 1)]
        t = b - a
        tangents.append(t.normalized() if t.length > 1e-9 else Vector((0, 0, 1)))
    verts, faces = [], []
    m = len(prof)
    for i, p in enumerate(pts):
        t = tangents[i]
        side = t.cross(crest_v)
        if side.length < 1e-6:
            side = Vector((1, 0, 0))
        side.normalize()
        up = crest_v.copy()
        u = i / (n - 1) if n > 1 else 0.0
        s = 1.0 if closed else taper(u, edge, floor)
        ang = twist * DEG * (u - 0.5)
        if abs(ang) > 1e-9:
            rot = Matrix.Rotation(ang, 3, t)
            side = (rot @ side).normalized()
            up = (rot @ up).normalized()
        for a, b in prof:
            verts.append(tuple(p + side * (a * half_w * s) + up * (-b * half_t * s)))
    ring = n if closed else n - 1
    for i in range(ring):
        j = (i + 1) % n
        for k in range(m):
            k2 = (k + 1) % m
            faces.append((i * m + k, i * m + k2, j * m + k2, j * m + k))
    if not closed:
        faces.append(tuple(range(m - 1, -1, -1)))
        base = (n - 1) * m
        faces.append(tuple(base + k for k in range(m)))
    ob = mesh_object(name, verts, faces, mat)
    if bevel_mat is not None:
        ob.data.materials.append(bevel_mat)
        top_i = len(ob.data.materials) - 1
        body = ring * m
        for poly in ob.data.polygons:
            if poly.index < body and poly.index % m in (0, 1, m - 2, m - 1):
                poly.material_index = top_i
    return ob


def local_arc(center_deg, span_deg, radius, steps, bow=0.0, z_bow=0.0, z0=0.0):
    """Arc in the local XY plane of an orbit rig; local +Z is the plane normal."""
    out = []
    for i in range(steps + 1):
        u = i / steps
        a = (center_deg + span_deg * (u - 0.5)) * DEG
        r = radius + bow * math.sin(math.pi * u)
        out.append(Vector((r * math.cos(a), r * math.sin(a),
                           z0 + z_bow * math.sin(math.pi * u))))
    return out


def bipyramid(name, radius, length, mat, sides=8, axis="Y", spin=0.0):
    verts, faces = [], []
    for k in range(sides):
        a = TAU * k / sides + spin * DEG
        if axis == "Z":
            verts.append((radius * math.cos(a), radius * math.sin(a), 0.0))
        else:
            verts.append((radius * math.cos(a), 0.0, radius * math.sin(a)))
    tip = (Vector((0.0, 0.0, length * 0.5)) if axis == "Z"
           else Vector((0.0, -length * 0.5, 0.0)))
    head = len(verts)
    verts.append(tuple(tip))
    tail = len(verts)
    verts.append(tuple(-tip))
    for k in range(sides):
        k2 = (k + 1) % sides
        faces.append((head, k, k2))
        faces.append((tail, k2, k))
    return mesh_object(name, verts, faces, mat)


def hinge(name, radius, depth, mat, sides=6):
    verts = []
    for z in (-depth * 0.5, depth * 0.5):
        for k in range(sides):
            a = TAU * k / sides
            verts.append((radius * math.cos(a), radius * math.sin(a), z))
    faces = [tuple(range(sides - 1, -1, -1)), tuple(range(sides, sides * 2))]
    for k in range(sides):
        k2 = (k + 1) % sides
        faces.append((k, k2, sides + k2, sides + k))
    return mesh_object(name, verts, faces, mat)


def cut_crystal(name, radius, length, mat, sides=6):
    verts = []
    for z, r in ((-length * 0.19, radius * 0.82),
                 (length * 0.19, radius * 0.82)):
        for k in range(sides):
            a = TAU * k / sides + 30.0 * DEG
            verts.append((r * math.cos(a), r * math.sin(a), z))
    lower = len(verts)
    verts.append((0.0, 0.0, -length * 0.5))
    upper = len(verts)
    verts.append((0.0, 0.0, length * 0.5))
    faces = []
    for k in range(sides):
        k2 = (k + 1) % sides
        faces.extend(((lower, k2, k), (k, k2, sides + k2, sides + k),
                      (upper, sides + k, sides + k2)))
    ob = mesh_object(name, verts, faces, mat)
    ob.data.materials.append(CORE_MID)
    ob.data.materials.append(CORE_DARK)
    for poly in ob.data.polygons:
        facet = (poly.index // 3) % 3
        poly.material_index = (0, 1, 2)[facet]
    return ob

def pulse_point(k, center_deg, u):
    a = (center_deg + SPAN * (u - 0.5)) * DEG

def keys(ob, path, frames_values, interp="LINEAR"):
    bpy.context.preferences.edit.keyframe_new_interpolation_type = interp
    for frame, value in frames_values:
        setattr(ob, path, value)
        ob.keyframe_insert(data_path=path, frame=frame)


def spin_about(ob, axis, base_deg, net_deg, amp_deg, cycles, phase, steps=48):
    """Rotation about one local axis. net_deg is a whole multiple of the
    layer's symmetry and the sine runs whole cycles, so pose and velocity match
    at frame 1 and frame 97."""
    out = []
    for i in range(steps + 1):
        t = i / steps
        ang = (base_deg + net_deg * t + amp_deg *
               math.sin(TAU * cycles * t + phase)) * DEG
        e = [0.0, 0.0, 0.0]
        e[axis] = ang
        out.append((round(1 + (END - 1) * t), tuple(e)))
    keys(ob, "rotation_euler", out, "LINEAR")


def wobble_axis(ob, axis, base_deg, amp_deg, cycles, phase, steps=48):
    out = []
    for i in range(steps + 1):
        t = i / steps
        ang = (base_deg + amp_deg * math.sin(TAU * cycles * t + phase)) * DEG
        e = [0.0, 0.0, 0.0]
        e[axis] = ang
        out.append((round(1 + (END - 1) * t), tuple(e)))
    keys(ob, "rotation_euler", out, "LINEAR")


def periodic_scale(ob, profile, steps=48):
    out = []
    for i in range(steps + 1):
        t = i / steps
        v = profile(t % 1.0)
        s = v if isinstance(v, tuple) else (v, v, v)
        out.append((round(1 + (END - 1) * t), s))
    keys(ob, "scale", out, "LINEAR")

def charge_profile(u):
    if u < 0.68:
        return 1.0 + 0.38 * (u / 0.68) ** 1.6
    if u < 0.78:
        return 1.38 - 0.48 * ((u - 0.68) / 0.10)
    return 0.90 + 0.10 * ((u - 0.78) / 0.22)

def join_by_material():
    animated = {ob.name for ob in scene.objects
                if ob.animation_data and ob.animation_data.action}
    buckets = {}
    for ob in list(scene.objects):
        if ob.type != "MESH" or ob.name in animated:
            continue
        mat = ob.data.materials[0].name if ob.data.materials else ""
        buckets.setdefault((ob.parent.name if ob.parent else "", mat), []).append(ob)
    merged = 0
    for (parent, mat), obs in buckets.items():
        if len(obs) < 2:
            continue
        bpy.ops.object.select_all(action="DESELECT")
        for ob in obs:
            ob.select_set(True)
        bpy.context.view_layer.objects.active = obs[0]
        bpy.ops.object.join()
        joined = bpy.context.view_layer.objects.active
        joined.name = f"{mat} batch"
        merged += 1
    print(f"JOIN merged {merged} material batches, "
          f"mesh objects now {sum(1 for o in scene.objects if o.type == 'MESH')}")

def framing_report():
    from bpy_extras.object_utils import world_to_camera_view
    deps = bpy.context.evaluated_depsgraph_get()
    worst = (0, 0.0, 0.0)
    for frame in (1, 13, 25, 37, 49, 61, 73, 85, 96):
        scene.frame_set(frame)
        deps.update()
        xmin = ymin = 9e9
        xmax = ymax = -9e9
        dmin, dmax = 9e9, -9e9
        for ob in scene.objects:
            if ob.type != "MESH":
                continue
            ev = ob.evaluated_get(deps)
            me = ev.to_mesh()
            mw = ev.matrix_world
            for v in me.vertices:
                co = mw @ v.co
                ndc = world_to_camera_view(scene, camera, co)
                xmin, xmax = min(xmin, ndc.x), max(xmax, ndc.x)
                ymin, ymax = min(ymin, ndc.y), max(ymax, ndc.y)
                dmin, dmax = min(dmin, co.y), max(dmax, co.y)
            ev.to_mesh_clear()
        print(f"FRAME {frame:3d} ndc_x[{xmin:.3f},{xmax:.3f}] ndc_y[{ymin:.3f},{ymax:.3f}] "
              f"fill={(xmax - xmin):.3f}x{(ymax - ymin):.3f} depth_y[{dmin:.2f},{dmax:.2f}]")
        if xmax - xmin > worst[1]:
            worst = (frame, xmax - xmin, ymax - ymin)
    print(f"FRAME widest fill={worst[1]:.3f}x{worst[2]:.3f} at frame {worst[0]} "
          f"(safe if < 0.94)")
    scene.frame_set(POSTER_FRAME)

def extent_report():
    deps = bpy.context.evaluated_depsgraph_get()
    deps.update()
    lo = Vector((9e9, 9e9, 9e9))
    hi = Vector((-9e9, -9e9, -9e9))
    tris = 0
    for ob in scene.objects:
        if ob.type != "MESH":
            continue
        ev = ob.evaluated_get(deps)
        me = ev.to_mesh()
        me.calc_loop_triangles()
        tris += len(me.loop_triangles)
        for v in me.vertices:
            co = ev.matrix_world @ v.co
            for i in range(3):
                lo[i] = min(lo[i], co[i])
                hi[i] = max(hi[i], co[i])
        ev.to_mesh_clear()
    size = hi - lo
    print(f"SIZE x={size.x:.3f} y={size.y:.3f} z={size.z:.3f} "
          f"depth_ratio={size.y / size.x:.3f} tris={tris}")
    scene.frame_set(POSTER_FRAME)


def verify_report():
    problems = []
    tex = 0
    for ob in scene.objects:
        if ob.type != "MESH":
            continue
        me = ob.data
        if me.uv_layers:
            problems.append(f"{ob.name}: UV layer")
        vol = 0.0
        for poly in me.polygons:
            vs = [me.vertices[i].co for i in poly.vertices]
            for i in range(1, len(vs) - 1):
                vol += vs[0].dot(vs[i].cross(vs[i + 1])) / 6.0
        if vol <= 0.0 and me.materials and me.materials[0].blend_method == "OPAQUE":
            problems.append(f"{ob.name}: volume {vol:.5f} (normals inward)")
    for mat in bpy.data.materials:
        if not mat.use_nodes:
            continue
        for nd in mat.node_tree.nodes:
            if nd.type == "TEX_IMAGE":
                tex += 1
    if tex:
        problems.append(f"{tex} image texture node(s)")
    print(f"VERIFY textures={tex} problems={len(problems)} {problems[:6]}")
    print(f"VERIFY materials={len(bpy.data.materials)}")

def report_glb():
    raw = GLB.read_bytes()
    length = struct.unpack("<I", raw[12:16])[0]
    doc = json.loads(raw[20:20 + length])
    acc = doc["accessors"]
    tris = 0
    for mesh in doc["meshes"]:
        for prim in mesh["primitives"]:
            if "indices" in prim:
                tris += acc[prim["indices"]]["count"] // 3
            else:
                tris += acc[prim["attributes"]["POSITION"]]["count"] // 3
    prims = sum(len(m["primitives"]) for m in doc["meshes"])
    with_normals = sum(1 for m in doc["meshes"] for p in m["primitives"]
                       if "NORMAL" in p["attributes"])
    anims = doc.get("animations", [])
    durations = [max(acc[s["input"]]["max"][0] for s in a["samplers"]
                     if acc[s["input"]].get("max")) for a in anims]
    print(f"GLB bytes={len(raw)} meshes={len(doc['meshes'])} nodes={len(doc['nodes'])} "
          f"prims={prims} tris={tris} materials={len(doc.get('materials', []))}")
    print(f"GLB animations={len(anims)} names={[a.get('name') for a in anims]} "
          f"durations={[round(d, 4) for d in durations]}")
    print(f"GLB normals={with_normals}/{prims} images={len(doc.get('images', []))} "
          f"uvs={sum(1 for m in doc['meshes'] for p in m['primitives'] if 'TEXCOORD_0' in p['attributes'])} "
          f"ext={doc.get('extensionsUsed')}")
    blended = [m.get("name") for m in doc.get("materials", [])
               if m.get("alphaMode") == "BLEND"]
    print(f"GLB blended={blended}")
